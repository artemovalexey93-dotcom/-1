            } else {

        // Расчет аннуитетного платежа
        function calculateAnnuityPayment(loanAmount, annualRate, months) {
            if (months === 0 || annualRate === 0 || loanAmount === 0) return 0;
            
            const monthlyRate = annualRate / 100 / 12;
            const annuityCoefficient = monthlyRate * Math.pow(1 + monthlyRate, months) / (Math.pow(1 + monthlyRate, months) - 1);
            
            return loanAmount * annuityCoefficient;
        }
        
        // Расчет графика платежей с досрочными платежами
        function calculateWithEarlyPayments(loanAmount, annualRate, months, monthlyPayment, earlyPayments, earlyType) {
            if (earlyPayments.length === 0) {
                return {
                    newMonthlyPayment: monthlyPayment,
                    newMonths: months,
                    totalInterest: monthlyPayment * months - loanAmount,
                    totalEarlyPayments: 0,
                    interestSavings: 0,
                    netSavings: 0
                };
            }
            
            const monthlyRate = annualRate / 100 / 12;
            let balance = loanAmount;
            let totalInterest = 0;
            let totalPaid = 0;
            let monthsPaid = 0;
            let currentPayment = monthlyPayment;
            let earlyPaymentIndex = 0;
            
            // Создаем копию досрочных платежей для работы
            const sortedPayments = [...earlyPayments].sort((a, b) => a.month - b.month);
            
            // Расчет для уменьшения срока
            if (earlyType === 'reduceTerm') {
                for (let month = 1; month <= months * 2; month++) { // Удваиваем лимит для случая сильного сокращения срока
                    if (balance <= 0.01) break;
                    
                    // Начисление процентов
                    const interest = balance * monthlyRate;
                    totalInterest += interest;
                    
                    // Обычный платеж
                    const principal = currentPayment - interest;
                    balance -= principal;
                    totalPaid += currentPayment;
                    monthsPaid++;
                    
                    // Проверяем досрочный платеж в этом месяце
                    if (earlyPaymentIndex < sortedPayments.length && sortedPayments[earlyPaymentIndex].month === month) {
                        const earlyAmount = sortedPayments[earlyPaymentIndex].amount;
                        balance -= earlyAmount;
                        totalPaid += earlyAmount;
                        earlyPaymentIndex++;
                        
                        // Пересчет срока после досрочного платежа
                        if (balance > 0) {
                            // Рассчитываем новый срок для оставшейся суммы с текущим платежом
                            const newMonths = Math.ceil(Math.log(currentPayment / (currentPayment - balance * monthlyRate)) / Math.log(1 + monthlyRate));
                            // Обновляем общее количество месяцев
                            months = month + newMonths;
                        }
                    }
                    
                    if (balance <= 0.01) {
                        // Корректировка при переплате
                        if (balance < 0) {
                            totalPaid += balance; // balance отрицательный
                        }
                        balance = 0;
                        break;
                    }
                }
                
                const originalTotalInterest = monthlyPayment * months - loanAmount;
                const interestSavings = originalTotalInterest - totalInterest;
                const totalEarlyPayments = sortedPayments.reduce((sum, p) => sum + p.amount, 0);
                const netSavings = interestSavings - totalEarlyPayments;
                
                return {
                    newMonthlyPayment: monthlyPayment,
                    newMonths: monthsPaid,
                    totalInterest,
                    totalEarlyPayments,
                    interestSavings,
                    netSavings: netSavings > 0 ? netSavings : 0
                };
            }
            // Расчет для уменьшения платежа
            else {
                let monthsPaid = 0;
                let totalInterest = 0;
                let totalEarlyPayments = 0;
                let currentBalance = loanAmount;
                let currentMonthlyPayment = monthlyPayment;
                
                for (let month = 1; month <= months; month++) {
                    if (currentBalance <= 0.01) break;
                    
                    // Начисление процентов
                    const interest = currentBalance * monthlyRate;
                    totalInterest += interest;
                    
                    // Обычный платеж
                    const principal = currentMonthlyPayment - interest;
                    currentBalance -= principal;
                    monthsPaid++;
                    
                    // Проверяем досрочный платеж в этом месяце
                    if (earlyPaymentIndex < sortedPayments.length && sortedPayments[earlyPaymentIndex].month === month) {
                        const earlyAmount = sortedPayments[earlyPaymentIndex].amount;
                        currentBalance -= earlyAmount;
                        totalEarlyPayments += earlyAmount;
                        earlyPaymentIndex++;
                        
                        // Пересчет платежа после досрочного платежа
                        if (currentBalance > 0 && month < months) {
                            const remainingMonths = months - month;
                            currentMonthlyPayment = calculateAnnuityPayment(currentBalance, annualRate, remainingMonths);
                        }
                    }
                    
                    if (currentBalance <= 0.01) {
                        if (currentBalance < 0) {
                            totalInterest += currentBalance; // Корректировка процентов
                        }
                        currentBalance = 0;
                        break;
                    }
                }
                
                const originalTotalInterest = monthlyPayment * months - loanAmount;
                const interestSavings = originalTotalInterest - totalInterest;
                const netSavings = interestSavings - totalEarlyPayments;
                
                return {
                    newMonthlyPayment: currentMonthlyPayment,
                    newMonths: months,
                    totalInterest,
                    totalEarlyPayments,
                    interestSavings,
                    netSavings: netSavings > 0 ? netSavings : 0
                };
            }
        }
        
        // Расчет налогового вычета
        function calculateTaxDeduction(propertyValue, totalInterest, annualIncome) {
            let ndflRate = ndfl13.checked ? 13 : 15;
            
            if (annualIncome > 5000000) {
                ndflRate = 15;
                ndfl15.checked = true;
                ndfl13.checked = false;
                ndfl15Option.classList.add('active');
                ndfl13Option.classList.remove('active');
            }
            
            const propertyDeductionBase = Math.min(propertyValue, PROPERTY_DEDUCTION_LIMIT);
            const propertyTaxDeduction = propertyDeductionBase * ndflRate / 100;
            
            const interestDeductionBase = Math.min(totalInterest, INTEREST_DEDUCTION_LIMIT);
            const interestTaxDeduction = interestDeductionBase * ndflRate / 100;
            
            const totalTaxDeduction = propertyTaxDeduction + interestTaxDeduction;
            
            const annualNdfl = annualIncome * ndflRate / 100;
            const annualNdflReturn = Math.min(totalTaxDeduction, annualNdfl);
            
            let yearsForFullReturn = 0;
            if (annualNdflReturn > 0) {
                yearsForFullReturn = Math.ceil(totalTaxDeduction / annualNdflReturn);
            }
            
            return {
                propertyTaxDeduction,
                interestTaxDeduction,
                totalTaxDeduction,
                annualNdflReturn,
                yearsForFullReturn,
                ndflRate
            };
        }
        
        // Основная функция расчета
        function calculateMortgage() {
            // Валидация
            const validation = validateInputs();
            
            if (!validation.isValid) {
                return;
            }
            
            const {propertyValue, downPayment, interestRate, totalMonths} = validation;
            
            // Годовой доход
            const annualIncome = parseFloat(annualIncomeInput.value) || 0;
            
            // Сумма кредита
            const loanAmount = propertyValue - downPayment;
            
            // Ежемесячный платеж
            const monthlyPayment = calculateAnnuityPayment(loanAmount, interestRate, totalMonths);
            
            // Общая сумма выплат
            const totalPayment = monthlyPayment * totalMonths;
            
            // Начисленные проценты
            const totalInterest = totalPayment - loanAmount;
            
            // Необходимый доход
            const requiredIncome = monthlyPayment * 2.5;
            
            // Расчет налогового вычета
            const taxDeduction = calculateTaxDeduction(propertyValue, totalInterest, annualIncome);
            
            // Расчет досрочных платежей
            let earlyResults = null;
            if (earlyPaymentCheckbox.checked) {
                const earlyPayments = collectEarlyPayments();
                const earlyType = reduceTerm.checked ? 'reduceTerm' : 'reducePayment';
                earlyResults = calculateWithEarlyPayments(loanAmount, interestRate, totalMonths, monthlyPayment, earlyPayments, earlyType);
                
                // Обновление блока результатов досрочного погашения
                newLoanTermElement.textContent = `${earlyResults.newMonths} ${getNoun(earlyResults.newMonths, 'месяц', 'месяца', 'месяцев')}`;
                newMonthlyPaymentElement.textContent = formatNumber(earlyResults.newMonthlyPayment, true);
                interestSavingsElement.textContent = formatNumber(earlyResults.interestSavings, true);
                totalEarlyPaymentsElement.textContent = formatNumber(earlyResults.totalEarlyPayments, true);
                netSavingsElement.textContent = formatNumber(earlyResults.netSavings, true);
                
                // Обновление пояснения
                if (earlyType === 'reduceTerm') {
                    earlyExplanation.innerHTML = `<strong>Уменьшение срока:</strong> Ежемесячный платеж остался прежним, срок кредита сократился на ${totalMonths - earlyResults.newMonths} месяцев. Вы сэкономили ${formatNumber(earlyResults.interestSavings, true)} на процентах.`;
                } else {
                    earlyExplanation.innerHTML = `<strong>Уменьшение платежа:</strong> Срок кредита не изменился, ежемесячный платеж уменьшился на ${formatNumber(monthlyPayment - earlyResults.newMonthlyPayment, true)}. Вы сэкономили ${formatNumber(earlyResults.interestSavings, true)} на процентах.`;
                }
            }
            
            // Обновление отображения результатов
            loanAmountElement.textContent = formatNumber(loanAmount, true);
            monthlyPaymentElement.textContent = formatNumber(monthlyPayment, true);
            totalInterestElement.textContent = formatNumber(totalInterest, true);
            totalPaymentElement.textContent = formatNumber(totalPayment, true);
            requiredIncomeElement.textContent = formatNumber(requiredIncome, true);
            
            // Обновление отображения налогового вычета
            propertyTaxDeductionElement.textContent = formatNumber(taxDeduction.propertyTaxDeduction, true);
            interestTaxDeductionElement.textContent = formatNumber(taxDeduction.interestTaxDeduction, true);
            totalTaxDeductionElement.textContent = formatNumber(taxDeduction.totalTaxDeduction, true);
            annualNdflReturnElement.textContent = formatNumber(taxDeduction.annualNdflReturn, true);
            yearsForFullReturnElement.textContent = taxDeduction.yearsForFullReturn;
            
            // Обновление отображения значений
            updateValueDisplays();
        }
        
        // Валидация ввода
        function validateInputs() {
            let isValid = true;
            
            // Сброс предыдущих ошибок
            propertyValueError.textContent = '';
            downPaymentError.textContent = '';
            interestRateError.textContent = '';
            loanTermError.textContent = '';
            
            propertyValueInput.classList.remove('invalid');
            downPaymentInput.classList.remove('invalid');
            interestRateInput.classList.remove('invalid');
            loanYearsSelect.classList.remove('invalid');
            loanMonthsSelect.classList.remove('invalid');
            
            // Проверка стоимости недвижимости
            const propertyValue = parseFloat(propertyValueInput.value);
            if (isNaN(propertyValue) || propertyValue <= 0) {
                propertyValueError.textContent = 'Введите положительное число';
                propertyValueInput.classList.add('invalid');
                isValid = false;
            } else if (propertyValue > 999999999) {
                propertyValueError.textContent = 'Максимальное значение 999,999,999';
                propertyValueInput.classList.add('invalid');
                isValid = false;
            }
            
            // Проверка первоначального взноса
            const downPayment = parseFloat(downPaymentInput.value);
            if (isNaN(downPayment) || downPayment < 0) {
                downPaymentError.textContent = 'Введите неотрицательное число';
                downPaymentInput.classList.add('invalid');
                isValid = false;
            } else if (downPayment > 99999999) {
                downPaymentError.textContent = 'Максимальное значение 99,999,999';
                downPaymentInput.classList.add('invalid');
                isValid = false;
            } else if (downPayment > propertyValue) {
                downPaymentError.textContent = 'Взнос не может превышать стоимость недвижимости';
                downPaymentInput.classList.add('invalid');
                isValid = false;
            }
            
            // Проверка процентной ставки
            const interestRate = parseFloat(interestRateInput.value);
            if (isNaN(interestRate) || interestRate <= 0) {
                interestRateError.textContent = 'Введите положительное число';
                interestRateInput.classList.add('invalid');
                isValid = false;
            } else if (interestRate > 30) {
                interestRateError.textContent = 'Максимальное значение 30%';
                interestRateInput.classList.add('invalid');
                isValid = false;
            }
            
            // Проверка срока кредита
            const years = parseInt(loanYearsSelect.value) || 0;
            const months = parseInt(loanMonthsSelect.value) || 0;
            const totalMonths = years * 12 + months;
            
            if (totalMonths <= 0) {
                loanTermError.textContent = 'Срок кредита должен быть больше 0 месяцев';
                loanYearsSelect.classList.add('invalid');
                loanMonthsSelect.classList.add('invalid');
                isValid = false;
            }
            
            // Проверка курса доллара
            if (currencyUsd.checked) {
                const newExchangeRate = parseFloat(exchangeRateInput.value);
                if (isNaN(newExchangeRate) || newExchangeRate <= 0) {
                    alert('Введите корректный курс доллара (положительное число)');
                    isValid = false;
                } else {
                    exchangeRate = newExchangeRate;
                    currentExchangeRateSpan.textContent = formatNumber(exchangeRate, true);
                }
            }
            
            return {isValid, propertyValue, downPayment, interestRate, years, months, totalMonths};
        }
